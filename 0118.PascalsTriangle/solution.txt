class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> outer;
        vector<int> inner;
        // iterate through rows
        for (int i = 0; i < numRows; ++i) {
            inner.clear();
            // 1st element
            inner.push_back(1);
            // iterate through middle elements (columns)
            int j;
            for (j = 1; j < i; ++j)
                inner.push_back(outer[i - 1][j - 1] + outer[i - 1][j]);
            // last element starting from 1st row
            if (i > 0)
                inner.push_back(1);
            outer.push_back(inner);
        }
        return outer;
    }
};
